// Node.js Express Backend for Safaricom Daraja STK Push
// Deployable to Vercel Serverless or Render Web Service

const express = require('express');
const axios = require('axios');
const app = express();
app.use(express.json());

const CONSUMER_KEY = process.env.DARAJA_CONSUMER_KEY;
const CONSUMER_SECRET = process.env.DARAJA_CONSUMER_SECRET;
const PASSKEY = process.env.DARAJA_PASSKEY;
const TILL_NUMBER = "7968544"; // Buy Goods Till Number

// OAuth Token Generation
async function getAccessToken() {
  const auth = Buffer.from(`${CONSUMER_KEY}:${CONSUMER_SECRET}`).toString('base64');
  const res = await axios.get('https://api.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials', {
    headers: { Authorization: `Basic ${auth}` }
  });
  return res.data.access_token;
}

// STK Push Request Route
app.post('/api/stkpush', async (req, res) => {
  try {
    const { phone, amount } = req.body;
    const token = await getAccessToken();
    const timestamp = new Date().toISOString().replace(/[^0-9]/g, '').slice(0, 14);
    const password = Buffer.from(`${TILL_NUMBER}${PASSKEY}${timestamp}`).toString('base64');

    const payload = {
      BusinessShortCode: TILL_NUMBER,
      Password: password,
      Timestamp: timestamp,
      TransactionType: "CustomerBuyGoodsOnline",
      Amount: amount || 150,
      PartyA: phone,
      PartyB: TILL_NUMBER,
      PhoneNumber: phone,
      CallBackURL: "https://your-domain.vercel.app/api/callback",
      AccountReference: "VeroraChat",
      TransactionDesc: "Activation Deposit"
    };

    const response = await axios.post('https://api.safaricom.co.ke/mpesa/stkpush/v1/processrequest', payload, {
      headers: { Authorization: `Bearer ${token}` }
    });

    res.json(response.data);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(3000, () => console.log('Verora Server running on port 3000'));

How to Host on Vercel, Render & Netlify

Publish this web app in under 2 minutes for free on popular static hosting platforms.

▲

Vercel Deployment

  1. Create a new repository on GitHub named verorachat.
  2. Upload this index.html file directly.
  3. Log into Vercel.com and click Add New Project.
  4. Import the GitHub repo and click Deploy (No build command needed).
🌐

Netlify Deployment

  1. Log into Netlify.com.
  2. Go to Sites tab and scroll to Drag and Drop section.
  3. Drag a folder containing this index.html file.
  4. Your site goes live instantly with SSL enabled!
⚡

Render Static Site

  1. Log into Render.com dashboard.
  2. Click New + and select Static Site.
  3. Connect your GitHub repository containing index.html.
  4. Leave Publish Directory as ./ and click Deploy.
